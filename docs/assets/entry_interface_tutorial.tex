\begin{frame}{What will we build?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{One datasource, several units}
        Create three in-memory run-to-failure units and compose their
        \texttt{CustomSingleSourceLoader}s by hand.
      \end{block}
      \begin{block}{Ragged by design}
        Keep every unit at its natural length---without padding or truncation.
      \end{block}
    \column{0.47\textwidth}
      \begin{block}{Two execution styles}
        Compare a classical \texttt{fit}/\texttt{predict} estimator with a
        deep-learning model trained on sliding windows.
      \end{block}
      \begin{block}{One public API}
        Preprocess, train, test, and receive metrics through
        \texttt{EntryInterface} in both cases.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{1. Create units with different lifetimes}
\begin{lstlisting}[style=python]
UNIT_LENGTHS = {
    "unit_short": 180,
    "unit_medium": 240,
    "unit_long": 300,
}

unit_frames = {
    unit_name: make_unit_frame(
        unit_index=unit_index,
        n_steps=n_steps,
        rng=rng,
    )
    for unit_index, (unit_name, n_steps)
    in enumerate(UNIT_LENGTHS.items())
}
\end{lstlisting}

  \begin{block}{Same schema, unequal time axes}
    Every frame contains \texttt{signal\_0}, \texttt{signal\_1},
    \texttt{signal\_2}, and \texttt{rul}; only the number of rows changes.
  \end{block}
\end{frame}

\begin{frame}[fragile]{2. Compose the units by hand}
\begin{lstlisting}[style=python]
from copy import deepcopy
from picid.interface import (
    CustomMultiSourceLoader,
    CustomSingleSourceLoader,
)

unit_loaders = {
    name: CustomSingleSourceLoader.load_from_csv(
        source=frame, target_column="rul", task_mode="rul",
        data_splitter=deepcopy(splitter_template), data_name=name,
    )
    for name, frame in unit_frames.items()
}

datasource = CustomMultiSourceLoader(
    sources=unit_loaders,
    task_mode="rul",
    data_name="manual_ragged_units",
)
\end{lstlisting}

  \small The dictionary key is the unit identity. Each unit receives its own
  splitter because splitters carry configuration and state.
\end{frame}

\begin{frame}{3. What is ragged data?}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{lrrr}
        \toprule
        \textbf{Unit} & \textbf{Train} & \textbf{Val} & \textbf{Test} \\
        \midrule
        short  & 108 & 48 & 48 \\
        medium & 144 & 60 & 60 \\
        long   & 180 & 72 & 72 \\
        \bottomrule
      \end{tabular}

      \vspace{0.8em}
      \texttt{[(108, 3), (144, 3), (180, 3)]}
    \column{0.42\textwidth}
      \begin{block}{Definition}
        Ragged data has a shared feature width but a different number of time
        steps per unit.
      \end{block}
      \begin{block}{PICID representation}
        A \texttt{list[np.ndarray]} preserves those unequal lengths. Calling
        \texttt{np.stack()} would fail unless we padded or truncated first.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{4. Build a shared preprocessed view}
\begin{lstlisting}[style=python]
def scaling_transforms():
    return [
        DataTransform(
            transform_name="scale_features",
            transform=MinMaxScalerSklearn(),
            metadata={"apply_to": "features", "fit_on": "train"},
        ),
        DataTransform(
            transform_name="scale_rul",
            transform=MinMaxScalerSklearn(),
            metadata={"apply_to": "rul", "fit_on": "train"},
        ),
    ]

deep_processed = interface.process_datasource(
    datasource, transforms=scaling_transforms(), cache=False
)
\end{lstlisting}

  \begin{block}{No leakage}
    Both scalers fit on the training split only. Fresh transform instances are
    created for each model-specific preprocessing run.
  \end{block}
\end{frame}

\begin{frame}[fragile]{5. Give each model the view it expects}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \textcolor{deepgreen}{\textbf{Deep-learning view}}
\begin{lstlisting}[style=python]
# One 2D array per unit
(time, features)

# Examples
(108, 3)
(144, 3)
(180, 3)
\end{lstlisting}
      Units stay separate and are windowed independently downstream.
    \column{0.50\textwidth}
      \textcolor{fitblue}{\textbf{Fit/predict view}}
\begin{lstlisting}[style=python]
DataTransform(
    transform_name="add_task_dim_features",
    transform=ReshapeTransform(
        "time feature -> 1 time feature"
    ),
    metadata={"apply_to": "features",
              "assign_to": "features"},
)
\end{lstlisting}
      The leading task dimension gives \texttt{(1, time, features)}; the
      tabular dataset then concatenates units without padding.
  \end{columns}
\end{frame}

\begin{frame}{6. Fit/predict versus deep learning}
  \centering
  \setlength{\tabcolsep}{6pt}
  \renewcommand{\arraystretch}{1.32}
  \begin{tabular}{p{0.20\textwidth}p{0.34\textwidth}p{0.34\textwidth}}
    \toprule
    \textbf{Aspect} & \textcolor{fitblue}{\textbf{Fit/predict}} &
    \textcolor{deepgreen}{\textbf{Deep learning}} \\
    \midrule
    Model & \texttt{xgboost\_fit\_predict} & \texttt{mlp} \\
    Input & Complete tabular task & Sliding windows per unit \\
    Training & \texttt{fit(X, y)} & Minibatches + backpropagation \\
    Ragged units & Concatenated, no padding & Windowed separately \\
    Tutorial budget & 25 estimators & One epoch \\
    \bottomrule
  \end{tabular}

  \vspace{1em}
  \begin{block}{The important comparison}
    The data adapter and optimisation loop differ, but the experiment-facing
    call remains \texttt{interface.train(...)}.
  \end{block}
\end{frame}

\begin{frame}[fragile]{7. Train the fit/predict model}
\begin{lstlisting}[style=python]
fit_predict_results = interface.train(
    run_name="entry_interface_fit_predict",
    model="xgboost_fit_predict",
    task_definition=task,
    datasource=fit_predict_processed,
    transforms=[],
    evaluators="default",
    loggers=[CsvLogger(name="entry_interface_fit_predict")],
    overrides=[
        "trainer.max_epochs=1",
        "trainer.accelerator=cpu",
        "trainer.devices=1",
        "datamodule.num_workers=0",
        "+model.n_estimators=25",
    ],
    enable_progress_bar=False,
    seed=7,
)
\end{lstlisting}

  \small Lightning provides orchestration here; the estimator itself follows
  the familiar \texttt{fit}/\texttt{predict} contract.
\end{frame}

\begin{frame}{One pipeline, two paths}
  \centering
  \begin{tabular}{c@{\hspace{0.7em}}c@{\hspace{0.7em}}c@{\hspace{0.7em}}c}
    \fbox{unit frames} & $\longrightarrow$ & \fbox{manual composition} &
    $\longrightarrow$ \\
  \end{tabular}

  \vspace{0.9em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Fit/predict branch}
        Scale $\rightarrow$ add task dimension $\rightarrow$ concatenate
        ragged units $\rightarrow$ \texttt{fit}/\texttt{predict}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Deep-learning branch}
        Scale $\rightarrow$ retain per-unit arrays $\rightarrow$ create
        windows $\rightarrow$ minibatch training
      \end{block}
  \end{columns}

  \vspace{0.7em}
  \fbox{same task definition, evaluators, loggers, and returned metric format}
\end{frame}
