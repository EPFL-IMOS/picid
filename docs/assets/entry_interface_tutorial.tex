\begin{frame}{Why use \texttt{EntryInterface}?}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \begin{block}{Python-facing API}
        Describe the experiment with ordinary Python objects: a datasource,
        transforms, a task definition, and a model name or config.
      \end{block}
      \begin{block}{Hydra underneath}
        The interface composes the internal configuration, creates the
        Lightning objects, trains the model, and runs evaluation.
      \end{block}
    \column{0.48\textwidth}
      \textbf{Tutorial outcome}
      \begin{enumerate}
        \item Generate a small synthetic RUL dataset.
        \item Split and scale it without external files.
        \item Train a built-in MLP for one CPU epoch.
        \item Receive the test metrics as Python data.
      \end{enumerate}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{1. Create a small datasource}
\begin{lstlisting}[style=python]
import numpy as np
import pandas as pd
from picid.data.preprocessing import TimeSplitter
from picid.interface import CustomSingleSourceLoader

rng = np.random.default_rng(7)
frame = pd.DataFrame(rng.normal(size=(240, 4)),
                     columns=["signal_0", "signal_1",
                              "signal_2", "rul"])

splitter = TimeSplitter(
    train=0.6, val=0.2, test=None, seq_len=12, pred_len=0,
    create_splits_for=["features", "timestamps", "rul"],
)
datasource = CustomSingleSourceLoader.load_from_csv(
    frame, target_column="rul", task_mode="rul",
    data_splitter=splitter, data_name="synthetic_rul",
)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{2. Preprocess once, then inspect or reuse}
\begin{lstlisting}[style=python]
from picid.interface import EntryInterface
from picid.transforms.base import DataTransform
from picid.transforms.base_transforms.scaler import MinMaxScalerSklearn

scale_features = DataTransform(
    transform_name="scale_features",
    transform=MinMaxScalerSklearn(),
    metadata={"apply_to": "features", "fit_on": "train"},
)

interface = EntryInterface()
processed = interface.process_datasource(
    datasource, transforms=[scale_features], cache=False
)

print(processed.data_dict.keys())  # train, val, test
\end{lstlisting}

\vspace{0.2em}
\begin{block}{Important}
The scaler is fitted only on the training split. Passing the resulting
\texttt{ProcessedDatasource} to multiple runs avoids repeating preprocessing.
\end{block}
\end{frame}

\begin{frame}[fragile]{3. Define the task and train}
\begin{lstlisting}[style=python]
from picid.interface.schemas.loggers import CsvLogger
from picid.interface.schemas.task_definition import Prognostic

task = Prognostic(task_type="rul", seq_len=12,
                  stride=4, stride_train=4)

results = interface.train(
    run_name="quick_entry_interface",
    model="mlp",
    task_definition=task,
    datasource=processed,
    transforms=[],                 # already applied above
    evaluators="default",
    loggers=[CsvLogger(name="quick_entry_interface")],
    overrides=["trainer.max_epochs=1",
               "trainer.accelerator=cpu",
               "datamodule.num_workers=0"],
    enable_progress_bar=False,
    seed=7,
)
print(results)
\end{lstlisting}
\end{frame}

\begin{frame}{What happens inside the call?}
  \centering
  \setlength{\tabcolsep}{7pt}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{cll}
    \toprule
    & \textbf{Stage} & \textbf{Result} \\
    \midrule
    1 & Compose configuration & Python inputs become a Hydra experiment \\
    2 & Build datasets & Windows follow the prognostics task definition \\
    3 & Instantiate model & \texttt{"mlp"} resolves to the built-in MLP config \\
    4 & Fit & Lightning trains for one epoch on CPU \\
    5 & Test & Evaluators return a list of metric dictionaries \\
    \bottomrule
  \end{tabular}

  \vspace{1em}
  \begin{block}{The key separation}
    \texttt{process\_datasource()} prepares data; \texttt{train()} composes and
    executes the experiment. Keeping the processed object makes both stages
    visible and easy to debug.
  \end{block}
\end{frame}
